---
// Archived when the site was reduced to Home and CV.
import { getCollection } from 'astro:content';
import BaseLayout from '../../layouts/BaseLayout.astro';
import PageHeader from '../../components/shared/PageHeader.astro';
import { withBase } from '../../utils/paths';

const notes = (await getCollection('notes'))
  .filter((entry) => !entry.data.draft)
  .sort((a, b) => b.data.date.valueOf() - a.data.date.valueOf());
---
<BaseLayout title="Notes — Hugo Yu" description="Notes on research, learning, and ordinary things worth keeping.">
  <PageHeader
    eyebrow="Longer thoughts"
    title="Notes"
    description="Not a feed. Just things that felt worth finishing and keeping."
  />

  <section class="shell narrow-shell note-index" aria-label="Notes">
    {notes.map((note) => (
      <article class="note-card">
        <time datetime={note.data.date.toISOString()}>
          {note.data.date.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })}
        </time>
        <div>
          <h2><a href={withBase(`/notes/${note.id}`)}>{note.data.title}</a></h2>
          <p>{note.data.description}</p>
          {note.data.tags.length > 0 && (
            <ul class="tag-list" aria-label="Tags">
              {note.data.tags.map((tag) => <li>{tag}</li>)}
            </ul>
          )}
        </div>
      </article>
    ))}
  </section>
</BaseLayout>
